\documentclass[border=8pt]{standalone}

\usepackage{amsmath}
\usepackage{tikz}
\usetikzlibrary{positioning,arrows.meta,shapes.geometric,shapes.misc}

% --- OpenTikZ palette (light / default, color-blind-friendly Okabe-Ito) ---
\definecolor{otblue}{HTML}{0072B2}
\definecolor{otorange}{HTML}{E69F00}
\definecolor{otteal}{HTML}{009E73}
\definecolor{otpurple}{HTML}{CC79A7}
\definecolor{otgray}{HTML}{5A5A5A}

\begin{document}
% ==== parameters =========================================================
\def\rowsep{0.72}
\def\branchxsep{1.35}
\def\boxminw{3.05}
\def\boxminh{0.90}
\def\startlabel{Predictive state $h_t$}
\def\readlabel{Estimate $u_t,\ r_\psi(h_t),\ p(E_t),\ \lambda_t^{(w)}$}
\def\checklabel{Are fine relations reliable?}
\def\proclabel{Keep micro-relational\\choices admissible}
\def\errorlabel{Mask micro-relational loss;\\continuous and macro remain admissible}
\def\scorelabel{Score all admissible joint pairs\\$(\Delta,\alpha)$ with prediction $+$ compute cost}
\def\endlabel{Select pair and predict to $t+\Delta_k$}
\def\yeslabel{high $r_\psi$}
\def\nolabel{low $r_\psi$}
\def\looplabel{next predictive state}
% =========================================================================

\begin{tikzpicture}[
    node distance=\rowsep cm,
    >={Stealth[length=2.4mm]},
    base/.style={
      line width=0.75pt, align=center, font=\sffamily\small,
      minimum width=\boxminw cm, minimum height=\boxminh cm,
      inner xsep=8pt, inner ysep=6pt,
    },
    terminal/.style={base, shape=rounded rectangle, draw=otteal!75!black, fill=otteal!15},
    process/.style={base, rounded corners=3pt, draw=otblue!75!black, fill=otblue!14},
    decision/.style={base, shape=diamond, aspect=2.65, inner sep=1pt,
      draw=otorange!85!black, fill=otorange!19, minimum width=4.1cm},
    masked/.style={process, draw=otpurple!80!black, fill=otpurple!14},
    score/.style={process, draw=otorange!80!black, fill=otorange!10, minimum width=5.0cm},
    flow/.style={draw=otgray!78, line width=0.9pt, ->},
    loop/.style={draw=otpurple!85!black, line width=1.0pt, ->},
    elabel/.style={font=\sffamily\footnotesize, text=otgray, fill=otgray!0, inner sep=1.5pt},
  ]

  \node[terminal] (start) {\startlabel};
  \node[process, below=of start] (read) {\readlabel};
  \node[decision, below=1.0cm of read] (check) {\checklabel};
  \node[process, below left=1.25cm and \branchxsep cm of check] (proc) {\proclabel};
  \node[masked, below right=1.25cm and \branchxsep cm of check] (error) {\errorlabel};
  \node[score, below=2.05cm of check] (scorepairs) {\scorelabel};
  \node[terminal, below=of scorepairs] (end) {\endlabel};

  \draw[flow] (start) -- (read);
  \draw[flow] (read) -- (check);
  \draw[flow] (check) -- (proc)
    node[midway, above left, elabel] {\yeslabel};
  \draw[flow] (check) -- (error)
    node[midway, above right, elabel] {\nolabel};
  \draw[flow] (proc) |- (scorepairs);
  \draw[flow] (error) |- (scorepairs);
  \draw[flow] (scorepairs) -- (end);
  \draw[loop] (end.west) -| ([xshift=-34mm]read.west) -- (read.west)
    node[pos=0.25, left, font=\sffamily\footnotesize, text=otpurple!85!black]
      {\looplabel};

\end{tikzpicture}
\end{document}
